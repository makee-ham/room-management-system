function templateSnapshotForSubmission(roomNo,submission,attempt=null) {
        if(submission?.templateSnapshot)return submission.templateSnapshot;
        if(submission?.templateId&&attempt?.templateSnapshot?.id===submission.templateId)return attempt.templateSnapshot;
        if(submission){
          const submittedTypeId=submission.roomMetaSnapshot?.typeId||attempt?.roomMetaSnapshot?.typeId||ROOMS.find(item=>item.no===roomNo)?.type||'standard',snapshotRoom=ROOMS.find(item=>item.type===submittedTypeId)?.no||roomNo;
          const snapshot=submission.kind==='퇴실 청소'?legacyCheckoutTemplateSnapshotFor(snapshotRoom):templateSnapshotFor(snapshotRoom,submission.kind||'퇴실 청소');
          if(!snapshot)return null;
          const submittedVersion=submission.templateVersion||snapshot.version;
          return submittedVersion===snapshot.version?snapshot:Object.freeze({...snapshot,version:submittedVersion});
        }
        return attempt?.templateSnapshot||templateSnapshotFor(roomNo,attempt?.kind||'퇴실 청소');
      }

function taskState(no) {
        const attemptId=state.currentAttemptByRoom?.[no]||null,attempt=attemptId?state.cleaningAttempts?.[attemptId]:null,snapshot=snapshotForAttempt(no,attempt);
        if(!state.taskInputs[no]) state.taskInputs[no]=createTaskInputsFromSnapshot(snapshot,{attemptId,checklist:{},uploads:[],note:'',candle:state.candles[no]||0});
        let task=state.taskInputs[no];
        if(snapshot&&(task.attemptId!==attemptId||task.templateId!==snapshot.id||task.templateVersion!==snapshot.version)){
          task=createTaskInputsFromSnapshot(snapshot,{...task,attemptId});state.taskInputs[no]=task;
        }
        const fixtures=['bed','bath','floor','supply'];
        if(!task.attemptId)task.attemptId=attemptId;
        task.uploads.forEach((upload,index)=>{if(!upload.image)upload.image={...demoUploadImageFixture(fixtures[index]||'bed')};});
        if(!task.issueDraft)task.issueDraft={type:'',note:'',photos:[]};
        if(!task.bombRoomDraft)task.bombRoomDraft={note:'',photos:[]};
        return task;
      }

function validatedSubmission(submission) {
        const attempt=submission?.attemptId?state.cleaningAttempts?.[submission.attemptId]||null:null;
        if(!submission||state.cleaningSubmissions?.[submission.id]!==submission||!attempt)return null;
        const lineageOk=attempt.id===submission.attemptId&&attempt.room===submission.room&&attempt.performerId===submission.performerId&&attempt.performerName===submission.performerName&&attempt.kind===submission.kind&&attempt.baseRateSnapshot===submission.baseRateSnapshot&&(attempt.reservationIdSnapshot||null)===(submission.reservationIdSnapshot||null)&&(attempt.guestCountSnapshot??null)===(submission.guestCountSnapshot??null);
        if(!lineageOk)return null;
        const attemptMeta=attempt.roomMetaSnapshot,submissionMeta=submission.roomMetaSnapshot;
        if(!attemptMeta||!submissionMeta||attemptMeta.roomNo!==submissionMeta.roomNo||attemptMeta.typeId!==submissionMeta.typeId||attemptMeta.typeName!==submissionMeta.typeName||attemptMeta.elevator!==submissionMeta.elevator)return null;
        const completedDate=timestampIsoDate(submission.completedAt,''),submittedDate=timestampIsoDate(submission.submittedAt,'');
        if(!completedDate||attempt.completedAt!==submission.completedAt||completedDate<attemptPlanDate(attempt,'')||!submittedDate||submittedDate<completedDate||weekStartIso(completedDate)!==submission.weekStart)return null;
        const attemptTemplate=attempt.templateSnapshot,submissionTemplate=submission.templateSnapshot;
        if(!attemptTemplate||!submissionTemplate||submission.templateId!==submissionTemplate.id||attemptTemplate.id!==submissionTemplate.id||attemptTemplate.version!==submissionTemplate.version||attemptTemplate.typeId!==submissionTemplate.typeId||attemptTemplate.kindId!==submissionTemplate.kindId||attemptTemplate.requirementsMode!==submissionTemplate.requirementsMode)return null;
        if(submissionTemplate.requirementsMode==='photo-only-v1'&&Object.keys(submission.checklist||{}).length)return null;
        const expectedPhotos=(submissionTemplate.photos||[]).map(item=>item.id).sort(),actualPhotos=(submission.uploads||[]).map(item=>item.id).sort();
        if(expectedPhotos.join('|')!==actualPhotos.join('|'))return null;
        if((submissionTemplate.photos||[]).some(item=>item.required&&submission.uploads?.find(upload=>upload.id===item.id)?.status!=='done'))return null;
        if((submission.uploads||[]).some(upload=>['failed','uploading'].includes(upload.status)))return null;
        if(!Array.isArray(submission.roomIssueIds)||!Array.isArray(submission.roomIssuesSnapshot)||!Number.isInteger(Number(submission.candleCountSnapshot))||Number(submission.candleCountSnapshot)<0)return null;
        const issueIds=submission.roomIssueIds.map(id=>String(id||'')),issueSnapshots=submission.roomIssuesSnapshot,snapshotIssueIds=issueSnapshots.map(record=>String(record?.id||''));
        if(issueIds.some(id=>!id)||snapshotIssueIds.some(id=>!id)||new Set(issueIds).size!==issueIds.length||new Set(snapshotIssueIds).size!==snapshotIssueIds.length||[...issueIds].sort().join('|')!==[...snapshotIssueIds].sort().join('|'))return null;
        const liveIssuesById=new Map(roomIssueRecords(submission.room).map(record=>[String(record.id),record]));
        const issueLineageOk=issueSnapshots.every(record=>{
          if(!record)return false;
          const live=liveIssuesById.get(String(record.id));
          if(!Array.isArray(record.photos)||!live||!Array.isArray(live.photos))return false;
          const snapshotPhotoIds=record.photos.map(photo=>String(photo?.id||'')),livePhotoIds=live.photos.map(photo=>String(photo?.id||'')),snapshotPhotoSignatures=record.photos.map(evidencePhotoSignature).sort(),livePhotoSignatures=live.photos.map(evidencePhotoSignature).sort();
          if(snapshotPhotoIds.some(id=>!id)||new Set(snapshotPhotoIds).size!==snapshotPhotoIds.length||[...snapshotPhotoIds].sort().join('|')!==[...livePhotoIds].sort().join('|')||snapshotPhotoSignatures.join('|')!==livePhotoSignatures.join('|'))return false;
          return record.room===submission.room&&record.attemptId===submission.attemptId&&record.submissionId===submission.id&&record.createdById===submission.performerId&&record.createdBy===submission.performerName&&live.room===submission.room&&live.attemptId===submission.attemptId&&live.submissionId===submission.id&&live.createdById===submission.performerId&&live.createdBy===submission.performerName;
        });
        if(!issueLineageOk)return null;
        if(submission.reportId){
          const reportSnapshot=submission.bombRoomReportSnapshot,liveReport=state.bombRoomReports?.[submission.reportId];
          if(!reportSnapshot||reportSnapshot.id!==submission.reportId||reportSnapshot.submissionId!==submission.id||reportSnapshot.attemptId!==submission.attemptId||reportSnapshot.room!==submission.room||reportSnapshot.reportedById!==submission.performerId||reportSnapshot.reportedBy!==submission.performerName||reportSnapshot.baseRateSnapshot!==submission.baseRateSnapshot||!Array.isArray(reportSnapshot.photos)||!reportSnapshot.photos.length)return null;
          if(!liveReport||liveReport.id!==submission.reportId||liveReport.submissionId!==submission.id||liveReport.attemptId!==submission.attemptId||liveReport.room!==submission.room||liveReport.reportedById!==submission.performerId||liveReport.reportedBy!==submission.performerName||liveReport.baseRateSnapshot!==submission.baseRateSnapshot||!Array.isArray(liveReport.photos)||!liveReport.photos.length)return null;
          const snapshotReportPhotoIds=reportSnapshot.photos.map(photo=>String(photo?.id||'')),liveReportPhotoIds=liveReport.photos.map(photo=>String(photo?.id||''));
          if(snapshotReportPhotoIds.some(id=>!id)||new Set(snapshotReportPhotoIds).size!==snapshotReportPhotoIds.length||[...snapshotReportPhotoIds].sort().join('|')!==[...liveReportPhotoIds].sort().join('|')||reportSnapshot.photos.map(evidencePhotoSignature).sort().join('|')!==liveReport.photos.map(evidencePhotoSignature).sort().join('|'))return null;
        }
        return submission;
      }

function applyHashParameters(params) {
        if(!params.has('scenario'))return false;
        state=makeScenario(Number(params.get('scenario'))||0);hydrateTemplateSnapshotsForState();
        if(['admin','maid'].includes(params.get('role')))state.role=params.get('role');
        if(state.detail&&!detailAllowedForRole(state.detail.type,state.role))state.detail=null;
        if(state.role==='admin'&&adminNav.some(n=>n.id===params.get('view')))state.adminView=params.get('view');
        if(state.role==='maid'&&maidNav.some(n=>n.id===params.get('view')))state.maidView=params.get('view');
        if(/^2026-08-(0[1-9]|[12]\d|3[01])$/.test(params.get('date')||''))applyOperationalDate(state,params.get('date'));
        if(['today','tomorrow'].includes(params.get('cleaningDay'))){state.cleaningTab=`assignment-${params.get('cleaningDay')}`;syncAssignmentDateForCleaningTab(state);}
        const requestedRoomFilter=['catalog','default'].includes(params.get('filter'))?'vacant':params.get('filter');
        if(['all','vacant','available','blocked','cleaning','occupied','checkout-inspection','extra-guests','candle','issues','early','late'].includes(requestedRoomFilter))state.roomFilter=requestedRoomFilter;
        if(['all',...Object.keys(ROOM_TYPES)].includes(params.get('type')))state.roomTypeFilter=params.get('type');
        if(['all',...Object.keys(ROOM_TYPES)].includes(params.get('assignmentType')))state.assignmentTypeFilter=params.get('assignmentType');
        if(/^\d{0,6}$/.test(params.get('q')||''))state.roomSearch=params.get('q')||'';
        const bookingAnchor=params.get('bookingAnchor')||'',legacyBookingMonth=params.get('bookingMonth')||'';
        if(/^\d{4}-(0[1-9]|1[0-2])-([0-2]\d|3[01])$/.test(bookingAnchor)){state.quickReservationAnchorDate=bookingAnchor;state.quickReservationFollowsToday=false;}
        else if(/^\d{4}-(0[1-9]|1[0-2])$/.test(legacyBookingMonth)){state.quickReservationAnchorDate=`${legacyBookingMonth}-15`;state.quickReservationFollowsToday=false;}
        else{state.quickReservationAnchorDate=DEMO_TODAY;state.quickReservationFollowsToday=true;}
        if(['all',...Object.keys(ROOM_TYPES)].includes(params.get('bookingType')))state.quickReservationType=params.get('bookingType');
        if(/^\d{0,6}$/.test(params.get('bookingQ')||''))state.quickReservationSearch=params.get('bookingQ')||'';
        const detailValue=params.get('detail');
        if(detailValue){state.detail=null;const separator=detailValue.indexOf(':'),type=separator>0?detailValue.slice(0,separator):'',id=separator>0?detailValue.slice(separator+1):'';if(detailAllowedForRole(type,state.role)&&id)state.detail={type,id,...(params.get('mode')?{mode:params.get('mode')}:{})};}
        if(state.cleaningTab==='assignment-today')activateNotifiedAssignmentsForDate(state.selectedDate);
        return true;
      }
